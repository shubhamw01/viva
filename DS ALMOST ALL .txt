#include <iostream>
#include <algorithm>
using namespace std;

struct Contact {
    string name;
    string phone;
    string address;
};

// Linear Search
int linearSearch(Contact c[], int n, string key) {
    for (int i = 0; i < n; i++) {
        if (c[i].name == key) {
            return i;
        }
    }
    return -1;
}

// Binary Search (contacts must be sorted by name)
int binarySearch(Contact c[], int n, string key) {
    int low = 0, high = n - 1;

    while (low <= high) {
        int mid = (low + high) / 2;

        if (c[mid].name == key) {
            return mid;
        }
        else if (c[mid].name < key) {
            low = mid + 1;
        }
        else {
            high = mid - 1;
        }
    }

    return -1;
}

// Display contact details
void displayContact(Contact c[], int index) {
    if (index == -1) {
        cout << "Contact not found\n";
    }
    else {
        cout << "\nContact found!\n";
        cout << "Name: " << c[index].name << endl;
        cout << "Phone: " << c[index].phone << endl;
        cout << "Address: " << c[index].address << endl;
    }
}

int main() {
    int n;
    cout << "Enter number of contacts: ";
    cin >> n;

    Contact c[100];

    for (int i = 0; i < n; i++) {
        cout << "\nEnter contact " << i + 1 << " details:\n";
        cout << "Name: ";
        cin >> c[i].name;
        cout << "Phone: ";
        cin >> c[i].phone;
        cout << "Address: ";
        cin >> c[i].address;
    }

    // Sort contacts by name for binary search
    sort(c, c + n, [](Contact a, Contact b) {
        return a.name < b.name;
    });

    string key;
    int choice, index;

    do {
        cout << "\n--- Phone Book ---\n";
        cout << "1. Linear Search\n";
        cout << "2. Binary Search\n";
        cout << "3. Exit\n";
        cout << "Enter choice: ";
        cin >> choice;

        switch (choice) {
            case 1:
                cout << "Enter name to search: ";
                cin >> key;
                index = linearSearch(c, n, key);
                displayContact(c, index);
                break;

            case 2:
                cout << "Enter name to search: ";
                cin >> key;
                index = binarySearch(c, n, key);
                displayContact(c, index);
                break;

            case 3:
                cout << "Exiting...\n";
                break;

            default:
                cout << "Invalid choice\n";
        }

    } while (choice != 3);

    return 0;
}

        cout << "\nEnter Roll No: ";
        cin >> s[i].rollNo;

        cout << "Enter Name: ";
        cin >> s[i].name;

        cout << "Enter Marks: ";
        cin >> s[i].marks;
    }
}

void displayStudents(Student s[], int n) {
    cout << "\nRoll No\tName\tMarks\n";
    for (int i = 0; i < n; i++) {
        cout << s[i].rollNo << "\t"
             << s[i].name << "\t"
             << s[i].marks << endl;
    }
}

void sortByMarks(Student s[],int n){
    for(int i = 0;i<n;i++){
        for(int j = 0;i<n - i - 1;j++){
            if(s[j].marks > s[j+1].marks){
                Student temp = s[j];
                s[j]=s[j+1];
                s[j+1] = temp;
            }
        }
    }
    cout << "Students sorted by marks (ascending).\n";
}

void sortByName(Student s[], int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (s[j].name > s[j + 1].name) {
                Student temp = s[j];
                s[j] = s[j + 1];
                s[j + 1] = temp;
            }
        }
    }
    cout << "Students sorted by name (A to Z).\n";
}

struct Node {
    Node* next;
    int productId;
    string name;
    int quantity;
    int price;
};

// Add Product
Node* addProduct(Node* head) {
    Node* newNode = new Node();

    cout << "Enter Product ID: ";
    cin >> newNode->productId;

    cout << "Enter Product Name: ";
    cin >> newNode->name;

    cout << "Enter Quantity: ";
    cin >> newNode->quantity;

    cout << "Enter Price: ";
    cin >> newNode->price;

    newNode->next = nullptr;

    if (head == nullptr) {
        head = newNode;
    }
    else {
        Node* temp = head;

        while (temp->next != nullptr) {
            temp = temp->next;
        }

        temp->next = newNode;
    }

    cout << "Product added successfully!\n";

    return head;
}

// Remove Product
Node* removeProduct(Node* head) {
    int id;

    cout << "Enter Product ID to remove: ";
    cin >> id;

    if (head == nullptr) {
        cout << "Inventory is empty!\n";
        return head;
    }

    if (head->productId == id) {
        Node* temp = head;
        head = head->next;
        delete temp;

        cout << "Product removed successfully!\n";
        return head;
    }

    Node* temp = head;

    while (temp->next != nullptr) {
        if (temp->next->productId == id) {
            Node* del = temp->next;
            temp->next = temp->next->next;

            delete del;

            cout << "Product removed successfully!\n";
            return head;
        }

        temp = temp->next;
    }

    cout << "Product not found!\n";

    return head;
}

// Display Products
void displayProducts(Node* head) {

struct Node {
    Node* next;
    Node* prev;
    string name;
};

// Insert Station
Node* insertNode(Node* head) {
    Node* newNode = new Node();

    newNode->next = nullptr;
    newNode->prev = nullptr;

    cout << "Enter station name: ";
    cin >> newNode->name;

    if (head == nullptr) {
        head = newNode;
    }
    else {
        Node* temp = head;

        while (temp->next != nullptr) {
            temp = temp->next;
        }

        temp->next = newNode;
        newNode->prev = temp;
    }

    cout << "Station inserted successfully!\n";

    return head;
}

// Delete Station
Node* deleteStation(Node* head) {
    if (head == nullptr) {
        cout << "Stations not found\n";
        return head;
    }

    string stationName;

    cout << "Enter station name to delete: ";
    cin >> stationName;

    Node* temp = head;

    while (temp != nullptr) {
        if (temp->name == stationName) {

            // Delete first station
            if (temp == head) {
                head = head->next;

                if (head != nullptr) {
                    head->prev = nullptr;
                }
            }

            // Delete middle or last station
            else {
                temp->prev->next = temp->next;

                if (temp->next != nullptr) {
                    temp->next->prev = temp->prev;
                }
            }

            delete temp;

            cout << "Station deleted successfully!\n";
            return head;
        }

        temp = temp->next;
    }

    cout << "Station not found!\n";

    return head;
}

#include <iostream>
using namespace std;

#define MAX 10

struct Customer{
    int id;
    string name;
};

Customer q[MAX];
int front = 0, rear = -1;

void enqueue(){
    if(rear == MAX-1){
        cout << "Queue Full";
        return;
    }
    rear++;
    cin >> q[rear].id >> q[rear].name;
    cout << "Customer Added";
}

void serve(){
    if(front > rear){
        cout << "No Customers";
        return;
    }
    cout << "Serving " << q[front].id << " " << q[front].name;
    front++;
}

void display(){
    if(front > rear){
        cout << "No Customers";
        return;
    }
    for(int i=front;i<=rear;i++){
        cout << "\n" << q[i].id << " " << q[i].name;
    }
}

void peek(){
    if(front > rear){
        cout << "No Customers";
        return;
    }
    cout << "Front: " << q[front].id << " " << q[front].name;
}

int main(){
    int choice;

    do {
        cout << "\n1. Enqueue Customer";
        cout << "\n2. Serve Customer";
        cout << "\n3. Display All";
        cout << "\n4. Front Customer";
        cout << "\n5. Exit";
        cout << "\nEnter Choice: ";
        cin >> choice;

        switch(choice) {
            case 1:
                enqueue();
                break;
            case 2:
                serve();
                break;
            case 3:
                display();
                break;
            case 4:
                peek();
                break;
            case 5:
                cout << "Exiting...";
                break;
            default:
                cout << "Invalid Choice";
        }
    } while(choice != 5);

    return 0;
}

using namespace std;
struct Node{
    int data;
    Node* left;
    Node* right;
};

Node* createNode(int value){
    Node* newNode = new Node;
    newNode -> data = value;
    newNode -> left = NULL;
    newNode->right = NULL;
    return newNode;
}

Node* insertNode(Node* root,int value){
    if(root == NULL){
        return createNode(value);
    }
    if(root->data < value){
        root->right = insertNode(root->right,value);
    }else if(root->data > value){
        root->left = insertNode(root->left,value);
    }else{
        cout<<"Duplicate entry not allowed";
    }
    return root;
}

void search(Node* root,int value){
    if(root == NULL){
        cout<<"Element not found";
    }else if(root->data == value){
        cout<<"Element found";
    }else if(root->data < value){
        search(root->right,value);
    }else if(root->data > value){
        search(root->left,value);
    }
}

void inorder(Node* root){
    if(root!= NULL){
        inorder(root->left);
        cout<<root->data;
        inorder(root->right);
    }
}

void preorder(Node* root){
    if(root!= NULL){
        cout<<root->data;
        preorder(root->left);
        preorder(root->right);
    }
}

void postorder(Node* root){
    if(root!= NULL){
        postorder(root->left);
        postorder(root->right);
        cout<<root->data<<" ";
    }
}

Node* findMin(Node* root){
    while (root->left != NULL) {
        root = root->left;
    }
    return root;
}

Node* deleteNode(Node* root,int value){
    if(root == NULL){
        cout<<"element not found";
    }

    if(value < root->data){
        root->left = deleteNode(root->left,value);
    }else if(value > root->data){
        root->right = deleteNode(root->right,value);


    }else{
        if(root->left == NULL && root->right == NULL){
            delete root;
            return NULL;

        }else if(root->left == NULL){
            Node* temp = root->right;
            delete root;
            return temp;
            
        }else{
            Node* temp = findMin(root->right);
            root->data = temp->data;
            root->right = deleteNode(root->right,temp->data);
        }
    }
    return root;
}

int depth(Node* root){
    if(root==NULL){
        return 0;
    }
    int leftDepth = depth(root->left);
    int rightDepth= depth(root->right);
    if(leftDepth>rightDepth){
        return leftDepth + 1;
    }else{
        return rightDepth + 1;
    }
}

Node* copyTree(Node* root){
    if(root == NULL){
        return NULL;
    }
    Node* newNode = createNode(root->data);
    newNode->left = copyTree(root->left);
    newNode->right = copyTree(root->right);
    return newNode;
}

#include <bits/stdc++.h>
using namespace std;

string text = "";

stack<string> st;

stack<string> redoStack;

void insert() {
    string word;

    cout << "Enter word: ";
    cin >> word;

    st.push(text);
    while (!redoStack.empty())
        redoStack.pop();

    text = text + word + " ";

    cout << "Text: " << text << endl;
}

void Delete() {
    if (text.empty()) {
        cout << "Text is empty" << endl;
        return;
    }

    st.push(text);
    while (!redoStack.empty())
        redoStack.pop();

    while (!text.empty() && text.back() == ' ')
        text.pop_back();

    while (!text.empty() && text.back() != ' ')
        text.pop_back();

    cout << "Text: " << text << endl;
}

void Undo() {
    if (st.empty()) {
        cout << "Nothing to undo" << endl;
        return;
    }

    redoStack.push(text);

    text = st.top();
    st.pop();

    cout << "Undo: " << text << endl;
}

void Redo() {
    if (redoStack.empty()) {
        cout << "Nothing to redo" << endl;
        return;
    }

    st.push(text);

    text = redoStack.top();
    redoStack.pop();

    cout << "Redo: " << text << endl;
}

int main() {
    int choice;

    do {
        cout << "\n1. Insert";
        cout << "\n2. Delete";
        cout << "\n3. Undo";
        cout << "\n4. Redo";
        cout << "\n5. Display";
        cout << "\n0. Exit";

        cout << "\nEnter choice: ";
        cin >> choice;

        switch (choice) {
            case 1: insert(); break;
            case 2: Delete(); break;
            case 3: Undo(); break;
            case 4: Redo(); break;
            case 5: cout << "Text: " << text << endl; break;

#include <iostream>
#include <string>
using namespace std;

#define size 10

struct Student {
    int id;
    string name;
    float marks;
    int used;   // 0 = empty, 1 = occupied, 2 = deleted
};

Student table[size];

// Hash function
int hashFunction(int id) {
    return id % size;
}

// Add student
void addStudent() {
    Student s;

    cout << "Enter student ID: ";
    cin >> s.id;

    cout << "Enter student name: ";
    cin >> s.name;

    cout << "Enter student marks: ";
    cin >> s.marks;

    int index = hashFunction(s.id);

    for (int i = 0; i < size; i++) {
        int pos = (index + i) % size;

        if (table[pos].used == 1 && table[pos].id == s.id) {
            cout << "Student ID already exists!\n";
            return;
        }

        if (table[pos].used == 0 || table[pos].used == 2) {
            table[pos] = s;
            table[pos].used = 1;
            cout << "Student added successfully!\n";
            return;
        }
    }

    cout << "Hash table is full!\n";
}

// Search student
void searchStudent() {
    int id;

    cout << "Enter student ID to search: ";
    cin >> id;

    int index = hashFunction(id);

    for (int i = 0; i < size; i++) {
        int pos = (index + i) % size;

        if (table[pos].used == 0) {
            break;
        }

        if (table[pos].used == 1 && table[pos].id == id) {
            cout << "\nStudent found!\n";
            cout << "ID: " << table[pos].id << endl;
            cout << "Name: " << table[pos].name << endl;
            cout << "Marks: " << table[pos].marks << endl;
            return;
        }
    }

    cout << "Student not found!\n";
}

// Delete student
void deleteStudent() {
    int id;

    cout << "Enter student ID to delete: ";
    cin >> id;

    int index = hashFunction(id);

    for (int i = 0; i < size; i++) {
        int pos = (index + i) % size;

        if (table[pos].used == 0) {
            break;
        }

        if (table[pos].used == 1 && table[pos].id == id) {
            table[pos].used = 2;
            cout << "Student deleted successfully!\n";
            return;
        }
    }

    cout << "Student not found!\n";
}

// Display all students
void displayStudents() {
    cout << "\n--- Student Records ---\n";

    for (int i = 0; i < size; i++) {
        if (table[i].used == 1) {
            cout << "ID: " << table[i].id
                 << " | Name: " << table[i].name
                 << " | Marks: " << table[i].marks << endl;
        }
    }
}

#include<iostream>
using namespace std;

int grades[4][4] = {
    {0, 0, 0, 0},
    {85, 0, 0, 0},
    {0, 0, 0, 92},
    {0, 0, 0, 0}
};

void calculateAvgGradeSub(){
    for(int j = 0; j < 4; j++){
        int sum = 0;
        int count = 0;

        for(int i = 0; i < 4; i++){
            if(grades[i][j] != 0){
                sum += grades[i][j];
                count++;
            }
        }

        if(count > 0){
            int avg = sum / count;
            cout << "Average marks: " << avg
                 << " Subject: " << j + 1 << endl;
        }
        else{
            cout << "No grades for Subject " << j + 1 << endl;
        }
    }
}

void highestGrade(){
    int highest = 0;

    for(int i = 0; i < 4; i++){
        for(int j = 0; j < 4; j++){
            if(grades[i][j] > highest){
                highest = grades[i][j];
            }
        }
    }

    cout << "Highest Grade: " << highest << endl;

    for(int i = 0; i < 4; i++){
        for(int j = 0; j < 4; j++){
            if(grades[i][j] == highest && highest != 0){
                cout << "Student with highest grade: "
                     << i + 1 << endl;
            }
        }
    }
}

void subWithHighestAvgGrade(){
    int maxAvg = 0;
    int subject = -1;

    for(int j = 0; j < 4; j++){
        int sum = 0;
        int count = 0;

        for(int i = 0; i < 4; i++){
            if(grades[i][j] != 0){
                sum += grades[i][j];
                count++;
            }
        }

        if(count > 0){
            int avg = sum / count;

            if(avg > maxAvg){
                maxAvg = avg;
                subject = j + 1;
            }
        }
    }

    if(subject != -1){
        cout << "Subject with highest average: "
             << subject << endl;
        cout << "Highest Average: " << maxAvg << endl;
    }
    else{
        cout << "No grades available" << endl;
    }
}

#include <bits/stdc++.h>
using namespace std;

// Operator Precedence
int precedence(char op) {
    if (op == '^') {
        return 3;
    }
    else if (op == '*' || op == '/') {
        return 2;
    }
    else if (op == '+' || op == '-') {
        return 1;
    }
    else {
        return 0;
    }
}

// Check Operator
bool isOperator(char ch) {
    return ch == '+' || ch == '-' || ch == '*' ||
           ch == '/' || ch == '^';
}

// Infix to Postfix
string infixToPostfix(string infix) {
    stack<char> s;
    string postfix = "";

    for (int i = 0; i < infix.length(); i++) {
        char ch = infix[i];

        if (ch == ' ') {
            continue;
        }

        if (isalnum(ch)) {
            postfix += ch;
        }
        else if (ch == '(') {
            s.push(ch);
        }
        else if (ch == ')') {
            while (!s.empty() && s.top() != '(') {
                postfix += s.top();
                s.pop();
            }

            if (!s.empty()) {
                s.pop();
            }
        }
        else if (isOperator(ch)) {
            while (!s.empty() && s.top() != '(' &&
                   (precedence(s.top()) > precedence(ch) ||
                   (precedence(s.top()) == precedence(ch) && ch != '^'))) {

                postfix += s.top();
                s.pop();
            }

            s.push(ch);
        }
    }

    while (!s.empty()) {
        postfix += s.top();
        s.pop();
    }

    return postfix;
}

// Infix to Prefix
string infixToPrefix(string infix) {

    // Reverse infix expression
    reverse(infix.begin(), infix.end());

    // Swap brackets
    for (int i = 0; i < infix.length(); i++) {

        if (infix[i] == '(') {
            infix[i] = ')';
        }
        else if (infix[i] == ')') {
            infix[i] = '(';
        }
    }

    // Convert to postfix
    string prefix = infixToPostfix(infix);

    // Reverse postfix to get prefix
    reverse(prefix.begin(), prefix.end());

    return prefix;
}

#include <iostream>
using namespace std;

struct Node{
    int data;
    int height;
    Node* left;
    Node* right;
};

int height(Node* n){
    if(n == NULL) return 0;
    return n->height;
}

int balanceFactor(Node* n){
    if(n == NULL) return 0;
    return height(n->left) - height(n->right);
}

void updateHeight(Node* n){
    n->height = 1 + max(height(n->left), height(n->right));
}

Node* rightRotate(Node* y){
    Node* x = y->left;
    y->left = x->right;
    x->right = y;
    updateHeight(y);
    updateHeight(x);
    return x;
}

Node* leftRotate(Node* x){
    Node* y = x->right;
    x->right = y->left;
    y->left = x;
    updateHeight(x);
    updateHeight(y);
    return y;
}

Node* balance(Node* n){
    updateHeight(n);
    int bf = balanceFactor(n);

    if(bf > 1){                                  // left heavy
        if(balanceFactor(n->left) < 0)
            n->left = leftRotate(n->left);       // LR case
        return rightRotate(n);                   // LL case
    }
    if(bf < -1){                                 // right heavy
        if(balanceFactor(n->right) > 0)
            n->right = rightRotate(n->right);    // RL case
        return leftRotate(n);                    // RR case
    }
    return n;
}

Node* insert(Node* root, int data){
    if(root == NULL){
        Node* n = new Node;
        n->data = data;
        n->height = 1;
        n->left = NULL;
        n->right = NULL;
        return n;
    }
    if(data < root->data)
        root->left = insert(root->left, data);
    else if(data > root->data)
        root->right = insert(root->right, data);
    else{
        cout << "Duplicate not allowed";
        return root;
    }
    return balance(root);
}

Node* del(Node* root, int data){
    if(root == NULL) return NULL;

    if(data < root->data)
        root->left = del(root->left, data);
    else if(data > root->data)
        root->right = del(root->right, data);
    else{
        if(root->left == NULL) return root->right;
        if(root->right == NULL) return root->left;

        Node* temp = root->right;
        while(temp->left != NULL) temp = temp->left;
        root->data = temp->data;
        root->right = del(root->right, temp->data);
    }
    return balance(root);
}

bool search(Node* root, int data){
    if(root == NULL) return false;
    if(data == root->data) return true;
    if(data < root->data) return search(root->left, data);
    return search(root->right, data);
}

}

void preorder(Node* root){
    if(root == NULL) return;

    preorder(root->left);
    preorder(root->right);
}
